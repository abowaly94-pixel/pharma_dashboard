REM ===================================
REM الأوامر المطلوبة للـ Deploy
REM نسخ ولصق في CMD
REM ===================================

REM 1. الذهاب لمجلد functions
cd functions

REM 2. حذف node_modules القديم
rmdir /s /q node_modules

REM 3. تثبيت Dependencies الجديدة
npm install

REM 4. Build الكود
npm run build

REM 5. Deploy على Firebase
npm run deploy

REM ===================================
REM بعد Deploy:
REM ===================================

REM 6. تفعيل FCM API من المتصفح:
REM https://console.cloud.google.com/apis/library/fcm.googleapis.com?project=pharmanow-754a7

REM 7. اختبار النظام:
REM افتح: http://localhost:5173
REM سجل دخول كـ Admin
REM اذهب إلى: /admin/notifications
REM أرسل إشعار تجريبي

REM 8. تحقق من Logs:
REM firebase functions:log
